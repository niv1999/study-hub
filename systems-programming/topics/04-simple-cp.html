0. משימה אחת, שלוש דרכים

code walkthrough של המרצה: אותה משימה — העתקת קובץ ממקור ליעד — כתובה שלוש פעמים, בשלוש שכבות, כדי לראות איפה נכנסת קריאת מערכת:

  • simple_cp.c — ספריית C (fopen / fread / fwrite / fclose), ניידת.
  • simple_cp_syscall.c — קריאות מערכת POSIX/לינוקס (open / read / write / close) — לב השיעור.

🎓 מהשיעור בכיתה (מהמחברת)

המשפט שהמרצה כתב מתחת לשתי הגרסאות הראשונות — ההגדרה של כל הקורס:

  • גרסה 1: "אותה תכנית — אבל משתמשת בפונקציות C";
  • גרסה 2: "תכנית מספר 2 — כתובה בקריאות שפונות לגרעין של מערכת ההפעלה".

שתי התוכניות עושות אותו דבר, אבל רק אחת פונה לגרעין במפורש. וליד גרסת ה-Windows הוא רשם "רק על Win 11" — המחיר של הירידה לשכבת הקרנל הוא אובדן ניידות.

מונחי בסיס: מעטפת (shell) — Bash מריצה בשבילכם תוכניות מהטרמינל (סעיף 6); argv — מערך מילות שורת הפקודה, argc — מספרן, כולל שם התוכנית; חוצץ (buffer) — "דלי" בייטים זמני בין המקור ליעד; קריאת מערכת — בקשת שירות מהקרנל בכל מה שנוגע לחומרה.

לשלושתן אותו שלד: בדיקת ארגומנטים (argc != 3 — בדיוק שני שמות קבצים), פתיחת המקור, פתיחת/יצירת היעד, לולאת קריאה-כתיבה, וסגירה.

1. גרסה 1 — simple_cp.c: ספריית C הסטנדרטית

אין כאן אף קריאת מערכת גלויה — הכול דרך stdio, שעובדת עם FILE * (stream) ולא עם מספרי קבצים.

הכותרות וההצהרות

#include <stdio.h>
#include <stdlib.h>

#define BUFFER_SIZE 4096

int main(int argc, char *argv[]) {
    FILE *source_file, *dest_file;
    char buffer[BUFFER_SIZE];
    size_t bytes_read;

שני מצביעי FILE * (מקור ויעד), חוצץ בגודל 4096 בייטים, ומונה size_t bytes_read — שימו לב לטיפוס.

בדיקת הארגומנטים

    /* Check if the user provided the correct number of command-line arguments */
    if (argc != 3) {
        fprintf(stderr, "Usage: %s <source-file> <dest-file>\n", argv[0]);
        exit(EXIT_FAILURE);
    }

אם לא סופקו בדיוק מקור ויעד — שורת Usage ל-stderr (לא ל-stdout!) ויציאה עם EXIT_FAILURE.

פתיחת שני הקבצים

    /* Open the source file for reading in binary mode */
    source_file = fopen(argv[1], "rb");
    if (source_file == NULL) {
        perror("Error opening source file");
        exit(EXIT_FAILURE);
    }

    /* Open the destination file for writing in binary mode */
    dest_file = fopen(argv[2], "wb");
    if (dest_file == NULL) {
        perror("Error opening destination file");
        fclose(source_file);
        exit(EXIT_FAILURE);
    }

פתיחה במצב בינארי: "rb" לקריאה, "wb" לכתיבה. כישלון של fopen ⇐ NULL, ואז perror() מדפיסה הודעת שגיאה מערכתית. ואם פתיחת היעד נכשלה — קודם fclose(source_file) ורק אז יוצאים.

לולאת ההעתקה והסגירה

    /* Read from source and write to destination in chunks */
    while ((bytes_read = fread(buffer, 1, BUFFER_SIZE, source_file)) > 0) {
        if (fwrite(buffer, 1, bytes_read, dest_file) != bytes_read) {
            perror("Error writing to destination file");
            fclose(source_file);
            fclose(dest_file);
            exit(EXIT_FAILURE);
        }
    }

    /* Close the file streams */
    fclose(source_file);
    fclose(dest_file);

    return EXIT_SUCCESS;
}

fread קוראת עד 4096 בייטים ומחזירה כמה נקראו בפועל; fwrite כותבת בדיוק bytes_read בייטים — ערך החזרה אחר הוא שגיאת כתיבה: סוגרים את שני הקבצים ויוצאים. קריאה שמחזירה 0 = סוף הקובץ — סוגרים ומחזירים EXIT_SUCCESS.

המסר: פונקציית הספרייה מתלבשת על קריאת מערכת — כמו בשרשרת fprintf → fwrite → write() במודול 03. הגרסה הבאה מוותרת על העטיפה וקוראת ל-open() בעצמה.

2. גרסה 2 — simple_cp_syscall.c: קריאות המערכת עצמן

הגרסה החשובה ביותר בשיעור — ההערות בקוד הן של המרצה, והן מגדירות במדויק מה כל קריאת מערכת עושה.

הכותרות וההצהרות

#include <fcntl.h>    /* Contains system calls for file control (open) */
#include <unistd.h>   /* Contains system calls for POSIX API (read, write, close) */
#include <stdio.h>    /* Used here only for perror() and printing */
#include <stdlib.h>

#define BUFFER_SIZE 4096

int main(int argc, char *argv[]) {
    int src_fd, dest_fd;  /* File descriptors: integers representing open files */
    char buffer[BUFFER_SIZE];
    ssize_t bytes_read, bytes_written;

שלושה שינויים: הכותרות הן fcntl.h (ל-open) ו-unistd.h (ל-read / write / close), ו-stdio.h נשארת רק ל-perror() והדפסות; במקום FILE * מחזיקים int — File Descriptor, מספר שלם שמייצג קובץ פתוח; והמונים הם ssize_t — עם סימן, לעומת size_t של fread.

בדיקת הארגומנטים

זהה לגרסת ה-stdio, מילה במילה.

    /* Verify correct command-line arguments */
    if (argc != 3) {
        fprintf(stderr, "Usage: %s <source-file> <dest-file>\n", argv[0]);
        exit(EXIT_FAILURE);
    }

פתיחת המקור — ההגדרה ל-open()

    /*
     * open() is a system call. It tells the Kernel: "I want to access this file."
     * The Kernel checks permissions and returns a 'File Descriptor' (an integer).
     * O_RDONLY: Open the file for reading only.
     */
    src_fd = open(argv[1], O_RDONLY);
    if (src_fd == -1) {
        perror("Failed to open source file");
        exit(EXIT_FAILURE);
    }

ההגדרה מהשקף, מילה במילה: open() היא קריאת מערכת שאומרת לקרנל — "אני רוצה לגשת לקובץ הזה". הקרנל בודק הרשאות ומחזיר File Descriptor — מספר שלם. וכישלון מוחזר כ--1 (לא NULL! זה לא מצביע, זה מספר).

ה-fd הוא אינדקס בטבלה פרטית שהקרנל מנהל לכל תהליך; הקרנל בוחר את המקום הפנוי הראשון, ולכן ה-fd הראשון הוא בדרך כלל 3 (0, 1, 2 תפוסים בקלט/פלט הסטנדרטיים) — ולכן read / write / close מקבלות int ולא שם קובץ (מודול 05).

פתיחת/יצירת היעד — הדגלים וההרשאות

    /*
     * O_WRONLY: Open for writing.
     * O_CREAT: Create the file if it doesn't exist.
     * O_TRUNC: If the file exists, wipe its content (truncate to 0 length).
     * 0644: Permissions (owner can read/write, others can read).
     */
    dest_fd = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (dest_fd == -1) {
        perror("Failed to create destination file");
        close(src_fd);
        exit(EXIT_FAILURE);
    }

כאן open מקבלת שלושה פרמטרים: הנתיב, שילוב דגלים ב-OR ביטי, ומצב הרשאות:

דגל / פרמטרמשמעות (מההערות בקוד)
O_WRONLYפתיחה לכתיבה בלבד.
O_CREATיצירת הקובץ אם אינו קיים.
O_TRUNCאם הקובץ קיים — מחיקת תוכנו (truncate לאורך 0).
0644הרשאות: הבעלים קורא/כותב, אחרים קוראים בלבד.

0644 הוא מספר אוקטלי (בסיס 8 — האפס בהתחלה מסמן זאת); כל ספרה היא קבוצת הרשאות, והערכים נסכמים: 4 = קריאה, 2 = כתיבה, 1 = הרצה.

הספרהלמי היא שייכתמורכבת מ-מה מותר בפועל
6הבעלים (Owner) — מי שיצר את הקובץ4 + 2קריאה וכתיבה
4הקבוצה (Group)4קריאה בלבד
4כל השאר (Others)4קריאה בלבד

הארגומנט השלישי רלוונטי רק בשילוב עם O_CREAT — לפתיחת המקור עם O_RDONLY לא הועבר ארגומנט שלישי כלל (מודול 06). וגם כאן: אם יצירת היעד נכשלה — close(src_fd) לפני היציאה.

לולאת הקריאה-כתיבה

    /*
     * read() is a system call. It requests the Kernel to copy data from the
     * file into our buffer. The Kernel puts the process to sleep until the
     * disk/filesystem delivers the data.
     */
    while ((bytes_read = read(src_fd, buffer, BUFFER_SIZE)) > 0) {
        /*
         * write() is a system call. It asks the Kernel to take data from our
         * buffer and place it into the destination file.
         */
        bytes_written = write(dest_fd, buffer, bytes_read);
        if (bytes_written != bytes_read) {
            perror("Error writing data");
            close(src_fd);
            close(dest_fd);
            exit(EXIT_FAILURE);
        }
    }

ההערה על read() קריטית: הקרנל מרדים את התהליך (puts the process to sleep) עד שהדיסק/מערכת הקבצים מספקים את הנתונים — הוא לא "מסתובב וממתין". write() מבקשת מהקרנל לקחת נתונים מהחוצץ ולהניח אותם ביעד, ותמיד בודקים bytes_written != bytes_read.

הסגירה

    /*
     * close() is a system call. It informs the Kernel that we are done.
     * The Kernel can now free the resources associated with these file descriptors.
     */
    close(src_fd);
    close(dest_fd);

    return EXIT_SUCCESS;
}

close() מודיעה לקרנל שסיימנו — ומרגע זה הוא רשאי לשחרר את המשאבים המשויכים ל-file descriptors.

שאלה בסגנון הבחינה — האם הבדיקה הזאת מספיקה?

במטלה my_copy.c סטודנט טיפל בשגיאת פתיחה כך. האם הקטע נכון פונקציונלית? 

int src_fd = open(argv[1], O_RDONLY);
if (src_fd == -1) {
    print_error("Error opening source file");
    exit(1);
}

כן — הקטע נכון לחלוטין מבחינה פונקציונלית.

הקרנל בודק הרשאות ורק אם הכול תקין מחזיר fd, ולכן בדיקה בודדת של src_fd == -1 מכסה את שני מצבי הכישלון:

  • קובץ המקור לא קיים בנתיב שב-argv[1].
  • הקובץ קיים אך ההרשאות אינן מתירות לקרוא ממנו — הקרנל מסרב ומחזיר גם כאן -1.

אין צורך בבדיקה נפרדת ל"קיים?" ובנוספת ל"מותר לקרוא?" — -1 הוא התשובה המשותפת לשתיהן; זו המהות של בדיקת ערך ההחזרה של כל קריאת מערכת.

מקור: simple_cp.pdf (עמ' 2–3) · שאלה לדוגמה מבחינת המרצה

3. גרסה 3 — simple_cp_windows.c: אותה משימה ב-Windows

open / read / write / close הן API של POSIX/לינוקס; לחלונות עולם מקביל משלה — אותו רעיון, שמות וטיפוסים אחרים.

הצהרות ופתיחת הקבצים

#include <windows.h>
#include <stdio.h>

#define BUFFER_SIZE 4096

int main(int argc, char *argv[]) {
    HANDLE hSource, hDest;
    char buffer[BUFFER_SIZE];
    DWORD bytesRead, bytesWritten;

    if (argc != 3) {
        printf("Usage: %s <source-file> <dest-file>\n", argv[0]);
        return 1;
    }

    /*
     * CreateFile is the Windows equivalent of open().
     * It returns a HANDLE, which is an opaque pointer/index
     * managed by the Windows Object Manager.
     */
    hSource = CreateFileA(argv[1], GENERIC_READ, FILE_SHARE_READ, NULL,
                          OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);

    if (hSource == INVALID_HANDLE_VALUE) {
        printf("Could not open source file. Error: %lu\n", GetLastError());
        return 1;
    }

    /* Create/Truncate destination file */
    hDest = CreateFileA(argv[2], GENERIC_WRITE, 0, NULL,
                        CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);

    if (hDest == INVALID_HANDLE_VALUE) {
        printf("Could not create destination file. Error: %lu\n", GetLastError());
        CloseHandle(hSource);
        return 1;
    }

לפי ההערה בקוד: CreateFile היא המקבילה של open() בחלונות, אבל מחזירה HANDLE — מצביע/אינדקס אטום (opaque) שמנוהל על ידי ה-Windows Object Manager, לעומת ה-fd הלינוקסי שהוא מספר שלם גלוי. כישלון ⇐ INVALID_HANDLE_VALUE, ופרטי השגיאה מ-GetLastError() (ב-%lu) במקום perror. הדגלים ממלאים את תפקיד O_RDONLY / O_CREAT / O_TRUNC (ליעד 0 = ללא שיתוף), וכשל בפתיחת היעד ⇐ CloseHandle(hSource).

הלולאה והסגירה

    /* Read and Write loop using Windows API */
    while (ReadFile(hSource, buffer, BUFFER_SIZE, &bytesRead, NULL) && bytesRead > 0) {
        if (!WriteFile(hDest, buffer, bytesRead, &bytesWritten, NULL) || bytesWritten != bytesRead) {
            printf("Error writing to file. Error: %lu\n", GetLastError());
            break;
        }
    }

    /* Close handles to release kernel resources */
    CloseHandle(hSource);
    CloseHandle(hDest);

    return 0;
}

ReadFile מחזירה הצלחה/כישלון, ואת ספירת הבייטים היא ממלאת דרך מצביע (&bytesRead, מטיפוס DWORD). בשגיאת כתיבה הקוד עושה break אל CloseHandle, כך שה-handles משתחררים בכל מקרה.

מקור: simple_cp.pdf (עמ' 4–5)

העשרה — מי בסוף קורא למי? (stdio מעל syscalls)

בלינוקס fopen / fread ממומשות בתוך glibc, מחזיקות חוצץ פנימי ב-user space (Buffered I/O), ויורדות דרך wrapper routines אל read / write. היתרון: פחות מעברים לקרנל (כל קריאת מערכת עולה מעבר בין מצבי ריצה) וקוד נייד; היתרון של syscalls ישירים: שליטה מלאה.

🎓 מהשיעור בכיתה (מהמחברת)

ההבדל עמוק יותר משמות הפונקציות — הוא מתחיל באבסטרקציה: איך כל מערכת מייצגת תהליך וחוט:

 WindowsLinux
תהליך (process) מיוצג ב-struct בשם E-Process אין הפרדה: גם תהליך וגם חוט הם task, ומיוצגים באותו struct — task_struct
חוט (thread) מיוצג ב-struct נפרד בשם E-Thread

שאלת השוואה קלאסית: Windows — שני מבנים; Linux — מבנה אחד (מודול 07, 09).

מקור: simple_cp.pdf (עמ' 1–5)

5. מלכודות למבחן

read() לא מבטיחה חוצץ מלא. הקריאה האחרונה תחזיר פחות מ-4096 (שארית הקובץ), ולכן הכתיבה מעבירה bytes_read ולא BUFFER_SIZE — אחרת היה נכתב זבל לסוף הקובץ. Partial I/O — במודול 6.

לכל API ערך כישלון אחר — אל תערבבו. fopen ⇐ NULL; open ⇐ -1; CreateFileA ⇐ INVALID_HANDLE_VALUE (בדיקת open() מול NULL = באג קלאסי). גם כל כתיבה נבדקת מול bytes_read.

דליפת descriptors היא באג קלאסי. אם פתיחת היעד נכשלת אחרי שהמקור נפתח, כל שלוש הגרסאות סוגרות קודם את המקור (fclose / close / CloseHandle).

הדגשה מהכיתה (המרצה צייר לידה משולש אזהרה) — פקודה פנימית לא יוצרת תהליך. cd היא built-in, ולכן לא נוצר עבורה fork/exec: אילו רצה בתהליך בן, שינוי הספרייה היה מת עם הבן.

ssize_t ולא size_t. read / write מחזירות ssize_t (עם סימן), ו-fread מחזירה size_t. תנאי הלולאה > 0 עוצר גם על 0 (סוף קובץ) וגם על ערך שלילי.

6. מאחורי הקלעים של Bash — מ-ENTER ועד exec

מה מבצעת מעטפת ה-Bash מהקלדת ./example ולחיצת ENTER — חמישה שלבים שסדרם המדויק הוא חומר חובה למבחן.

1

קליטת השורה — Terminal Line Discipline

ה-TTY Driver שבקרנל מבצע Echo ואוגר את התווים ב-Input Buffer; רק ENTER (תו \n) מעביר את השורה ל-Bash.

2

פענוח השורה — Tokenization & Command Line Parsing

ניתוח לקסיקלי (Lexical Analysis) לתוך tokens. ./ מסמן נתיב יחסי (Relative Path) בתיקיית העבודה — ולא פקודה גלובלית שמחפשים ב-PATH.

3

פקודה פנימית? — Built-in Command Check

./example אינו תואם אף פקודה פנימית (cd, echo, export, pwd) ⇒ פקודה חיצונית (External Executable File) ⇒ נדרש תהליך חדש.

🎓 מהשיעור בכיתה (מהמחברת)

הרשימה שהמרצה נקב בה בעל פה לפקודות פנימיות הייתה אחרת מזו שבדף העזר: cd, alias, history — כדאי להכיר את שתי הרשימות; הפואנטה זהה: אלה פקודות שה-Bash מבצע בעצמו, בלי ליצור תהליך חדש. את השלבים הוא מנה בכיתה באותו סדר: קליטת ה-command string ← parsing ל-tokens ← ה-token הראשון הוא שם הפקודה ← בדיקה אם היא פנימית ← ואם לא — fork() ואחריו exec().

4

שכפול התהליך — קריאת המערכת fork()

עותק (Clone) של task_struct של ה-Bash: בן עם PID משלו, ושכפול של fd_array ומצבי הזיכרון. 0 חוזר לבן, וה-PID של הבן לאב — שאלת מבחן קלאסית.

5

הבן מפעיל את execve()

הבן מזהה את עצמו (קיבל 0 מ-fork) ורק אז קורא ל:

execve("/home/izhak/system programming/example", argv, envp)

הקרנל מאמת את ה-ELF Header (Executable and Linkable Format — "תעודת הזהות" של קובץ הרצה בלינוקס: מעיד שהוא תקין ומאיפה להתחיל), מוחק את מרחב הזיכרון הישן שהועתק ב-fork, וטוען למרחב חדש את הקטעים הבינאריים של example.

מכאן ה-Bash נשאר ברקע בהמתנה (Sleep / Wait), ו-example רץ במרחב המשתמש.

התוכנית שהוקרנה בכיתה — fork / execlp / wait

🎓 מהשיעור בכיתה (מהמחברת)

המרצה סגר את חמשת השלבים בהקרנת קטע קוד שמדגים את שלבים 4–5:

#include <stdio.h>
void main(int argc, char *argv[])
{
  int pid ;

      /* fork another process */
      pid = fork();

      if (pid < 0) { /* error occurred */
         fprintf(stderr, "Fork Failed");
         exit (-1);
      }
      else if (pid == 0) {   /* child process */
         execlp("/bin/ls", "ls", NULL);
      }
      else {   /* parent process */
         wait(NULL);   /* parent will wait for the child to complete */
         printf("Child Complete");
         exit(0);
      }
}

שלושת הענפים של ה-if הם שלוש התוצאות של fork():

  • pid < 0 — כישלון: מדפיסים ל-stderr ויוצאים.
  • pid == 0 — הבן: execlp("/bin/ls", "ls", NULL) מחליף את התוכנית שרצה בתוכו ב-ls — שלב 5, בכתיבה שלנו במקום ה-Bash.
  • אחרת — האב: wait(NULL) עוצר אותו עד שהבן מסיים, ורק אז מודפס "Child Complete" — ולכן שורת הפקודה חוזרת רק כשהתוכנית נגמרת (מודול 07).
🎓 מהשיעור בכיתה (מהמחברת)

תהליך הוא "יקר", חוט הוא "זול". בכיתה המרצה העמיד זה מול זה שני קטעי קוד — אחד שיוצר עבודה חדשה ב-fork() ואחד (שרת מקבילי) ב-pthread_create(…) — והדגיש ש-fork יקר בהרבה: הוא מייצר task חדש עם מרחב זיכרון משלו, בעוד pthread_create יוצר חוט בתוך אותו תהליך. לכן שרתים שמטפלים בהרבה לקוחות בו-זמנית עובדים לרוב בחוטים (מודול 09).
