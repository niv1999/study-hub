⏱ זמן משוער: 50 דק'

0. משימה אחת, שלוש דרכים

בשיעור הקודם למדנו מה זו קריאת מערכת ואיך היא עובדת מתחת למכסה המנוע. עכשיו רואים את זה בקוד אמיתי. המסמך של המרצה הוא code walkthrough: אותה משימה בדיוק — להעתיק קובץ ממקור ליעד (cp פשוט) — כתובה שלוש פעמים, בשלוש שכבות שונות של המערכת:

  • simple_cp.c — עם ספריית C הסטנדרטית (stdio: fopen / fread / fwrite / fclose) — ניידת וגבוהה יותר.
  • simple_cp_syscall.c — עם קריאות המערכת של POSIX/לינוקס (open / read / write / close) — לב השיעור.
  • simple_cp_windows.c — עם ה-Windows API (CreateFileA / ReadFile / WriteFile / CloseHandle) — להשוואה.

לשלוש הגרסאות יש בדיוק אותו שלד: בדיקה ש-argc != 3 (כלומר שהמשתמש סיפק בדיוק שני ארגומנטים: argv[0] הוא שם התוכנית, argv[1] קובץ המקור, argv[2] קובץ היעד), פתיחת המקור, פתיחת/יצירת היעד, לולאת קריאה-כתיבה, וסגירה. ובשלושתן מופיעה אותה שורה: #define BUFFER_SIZE 4096 — ההעתקה נעשית במקטעים (chunks) של 4KB, לא בייט-בייט.

🆕 חדש בעמוד — דף העזר של המרצה על Bash, fork() ו-exec() עלה למודל: בסעיף 6 בהמשך העמוד נעקוב, שלב אחר שלב, אחרי כל מה שקורה מרגע ההקלדה של ./example ועד שהתוכנית רצה.

מקור: simple_cp.pdf (עמ' 1–5)

1. גרסה 1 — simple_cp.c: ספריית C הסטנדרטית

נקרא את הגרסה הראשונה שורה-אחר-שורה. כאן עוד אין אף קריאת מערכת גלויה — הכול עובר דרך stdio, שעובדת עם FILE * (stream) במקום עם מספרי קבצים.

הכותרות וההצהרות

#include <stdio.h>
#include <stdlib.h>

#define BUFFER_SIZE 4096

int main(int argc, char *argv[]) {
    FILE *source_file, *dest_file;
    char buffer[BUFFER_SIZE];
    size_t bytes_read;

שני מצביעי FILE * — אחד למקור ואחד ליעד; חוצץ (buffer) בגודל 4096 בייטים; ומונה size_t bytes_read — שימו לב לטיפוס, נחזור אליו בגרסה הבאה.

בדיקת הארגומנטים

    /* Check if the user provided the correct number of command-line arguments */
    if (argc != 3) {
        fprintf(stderr, "Usage: %s <source-file> <dest-file>\n", argv[0]);
        exit(EXIT_FAILURE);
    }

אם המשתמש לא סיפק בדיוק מקור ויעד — מדפיסים שורת Usage ל-stderr (לא ל-stdout!) ויוצאים עם EXIT_FAILURE.

פתיחת שני הקבצים

    /* Open the source file for reading in binary mode */
    source_file = fopen(argv[1], "rb");
    if (source_file == NULL) {
        perror("Error opening source file");
        exit(EXIT_FAILURE);
    }

    /* Open the destination file for writing in binary mode */
    dest_file = fopen(argv[2], "wb");
    if (dest_file == NULL) {
        perror("Error opening destination file");
        fclose(source_file);
        exit(EXIT_FAILURE);
    }

הפתיחה נעשית במצב בינארי: "rb" לקריאה, "wb" לכתיבה. כישלון של fopen מוחזר כ-NULL, ואז perror() מדפיסה הודעת שגיאה מערכתית. ושימו לב לפרט הקטן שהוא ענק: אם פתיחת היעד נכשלה — קודם fclose(source_file), ורק אחר כך יוצאים. המקור כבר פתוח, ואסור להשאיר אותו מאחור.

לולאת ההעתקה והסגירה

    /* Read from source and write to destination in chunks */
    while ((bytes_read = fread(buffer, 1, BUFFER_SIZE, source_file)) > 0) {
        if (fwrite(buffer, 1, bytes_read, dest_file) != bytes_read) {
            perror("Error writing to destination file");
            fclose(source_file);
            fclose(dest_file);
            exit(EXIT_FAILURE);
        }
    }

    /* Close the file streams */
    fclose(source_file);
    fclose(dest_file);

    return EXIT_SUCCESS;
}

fread קוראת עד 4096 בייטים ומחזירה כמה נקראו בפועל; fwrite כותבת בדיוק bytes_read בייטים — ואם היא החזירה מספר אחר, זו שגיאת כתיבה: סוגרים את שני הקבצים ויוצאים. כשהקריאה מחזירה 0, הקובץ נגמר — סוגרים ומחזירים EXIT_SUCCESS.

מקור: simple_cp.pdf (עמ' 1)

2. גרסה 2 — simple_cp_syscall.c: קריאות המערכת עצמן

עכשיו מורידים את שכבת ה-stdio ומדברים עם הקרנל ישירות. זו הגרסה החשובה ביותר בשיעור — ההערות בקוד הן של המרצה, והן מגדירות במדויק מה כל קריאת מערכת עושה.

הכותרות וההצהרות

#include <fcntl.h>    /* Contains system calls for file control (open) */
#include <unistd.h>   /* Contains system calls for POSIX API (read, write, close) */
#include <stdio.h>    /* Used here only for perror() and printing */
#include <stdlib.h>

#define BUFFER_SIZE 4096

int main(int argc, char *argv[]) {
    int src_fd, dest_fd;  /* File descriptors: integers representing open files */
    char buffer[BUFFER_SIZE];
    ssize_t bytes_read, bytes_written;

שלושה שינויים מהותיים לעומת הגרסה הקודמת: הכותרות הן fcntl.h (בשביל open) ו-unistd.h (בשביל read / write / close), כאשר stdio.h נשארת רק בשביל perror() והדפסות; במקום FILE * מחזיקים int פשוט — File Descriptor, מספר שלם שמייצג קובץ פתוח; והמונים הם מטיפוס ssize_t — טיפוס עם סימן, לעומת size_t של fread.

בדיקת הארגומנטים

    /* Verify correct command-line arguments */
    if (argc != 3) {
        fprintf(stderr, "Usage: %s <source-file> <dest-file>\n", argv[0]);
        exit(EXIT_FAILURE);
    }

זהה לחלוטין לגרסת ה-stdio — בדיקת קלט היא בדיקת קלט בכל שכבה.

פתיחת המקור — ההגדרה של המרצה ל-open()

    /*
     * open() is a system call. It tells the Kernel: "I want to access this file."
     * The Kernel checks permissions and returns a 'File Descriptor' (an integer).
     * O_RDONLY: Open the file for reading only.
     */
    src_fd = open(argv[1], O_RDONLY);
    if (src_fd == -1) {
        perror("Failed to open source file");
        exit(EXIT_FAILURE);
    }

ההגדרה מהשקף, מילה במילה: open() היא קריאת מערכת שאומרת לקרנל — "אני רוצה לגשת לקובץ הזה". הקרנל בודק הרשאות ומחזיר File Descriptor — מספר שלם. וכישלון מוחזר כ--1 (לא NULL! זה לא מצביע, זה מספר).

פתיחת/יצירת היעד — הדגלים וההרשאות

    /*
     * O_WRONLY: Open for writing.
     * O_CREAT: Create the file if it doesn't exist.
     * O_TRUNC: If the file exists, wipe its content (truncate to 0 length).
     * 0644: Permissions (owner can read/write, others can read).
     */
    dest_fd = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (dest_fd == -1) {
        perror("Failed to create destination file");
        close(src_fd);
        exit(EXIT_FAILURE);
    }

כאן open מקבלת שלושה פרמטרים: הנתיב, שילוב דגלים ב-OR ביטי, ומצב הרשאות:

דגל / פרמטרמשמעות (מההערות בקוד)
O_WRONLYפתיחה לכתיבה בלבד.
O_CREATיצירת הקובץ אם אינו קיים.
O_TRUNCאם הקובץ קיים — מחיקת תוכנו (truncate לאורך 0).
0644הרשאות: הבעלים קורא/כותב, אחרים קוראים בלבד.

וגם כאן, אותו דפוס ניקוי: אם יצירת היעד נכשלה — close(src_fd) לפני היציאה.

לולאת הקריאה-כתיבה

    /*
     * read() is a system call. It requests the Kernel to copy data from the
     * file into our buffer. The Kernel puts the process to sleep until the
     * disk/filesystem delivers the data.
     */
    while ((bytes_read = read(src_fd, buffer, BUFFER_SIZE)) > 0) {
        /*
         * write() is a system call. It asks the Kernel to take data from our
         * buffer and place it into the destination file.
         */
        bytes_written = write(dest_fd, buffer, bytes_read);
        if (bytes_written != bytes_read) {
            perror("Error writing data");
            close(src_fd);
            close(dest_fd);
            exit(EXIT_FAILURE);
        }
    }

ההערה של המרצה על read() היא נקודה קריטית: הקרנל מרדים את התהליך (puts the process to sleep) עד שהדיסק/מערכת הקבצים מספקים את הנתונים. התהליך לא "מסתובב וממתין" — הוא ישן, והקרנל יעיר אותו כשהנתונים מוכנים. write() מבקשת מהקרנל לקחת נתונים מהחוצץ שלנו ולהניח אותם בקובץ היעד — ותמיד בודקים bytes_written != bytes_read כטיפול בשגיאה.

הסגירה

    /*
     * close() is a system call. It informs the Kernel that we are done.
     * The Kernel can now free the resources associated with these file descriptors.
     */
    close(src_fd);
    close(dest_fd);

    return EXIT_SUCCESS;
}

close() מודיעה לקרנל שסיימנו — ומרגע זה הקרנל רשאי לשחרר את המשאבים המשויכים ל-file descriptors האלה.

מקור: simple_cp.pdf (עמ' 2–3)

3. גרסה 3 — simple_cp_windows.c: אותה משימה ב-Windows

הגרסה השלישית מזכירה לנו ש-open / read / write / close הן API של POSIX/לינוקס — לחלונות יש עולם מקביל משלה, עם אותו רעיון בדיוק אבל שמות וטיפוסים אחרים.

הצהרות ופתיחת הקבצים

#include <windows.h>
#include <stdio.h>

#define BUFFER_SIZE 4096

int main(int argc, char *argv[]) {
    HANDLE hSource, hDest;
    char buffer[BUFFER_SIZE];
    DWORD bytesRead, bytesWritten;

    if (argc != 3) {
        printf("Usage: %s <source-file> <dest-file>\n", argv[0]);
        return 1;
    }

    /*
     * CreateFile is the Windows equivalent of open().
     * It returns a HANDLE, which is an opaque pointer/index
     * managed by the Windows Object Manager.
     */
    hSource = CreateFileA(argv[1], GENERIC_READ, FILE_SHARE_READ, NULL,
                          OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);

    if (hSource == INVALID_HANDLE_VALUE) {
        printf("Could not open source file. Error: %lu\n", GetLastError());
        return 1;
    }

    /* Create/Truncate destination file */
    hDest = CreateFileA(argv[2], GENERIC_WRITE, 0, NULL,
                        CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);

    if (hDest == INVALID_HANDLE_VALUE) {
        printf("Could not create destination file. Error: %lu\n", GetLastError());
        CloseHandle(hSource);
        return 1;
    }

ההערה בקוד אומרת הכול: CreateFile היא המקבילה של open() בחלונות, אבל היא מחזירה HANDLE — מצביע/אינדקס אטום (opaque) שמנוהל על ידי ה-Windows Object Manager. לעומת ה-fd הלינוקסי, שהוא סתם מספר שלם גלוי. כישלון מסומן בערך המיוחד INVALID_HANDLE_VALUE, ופרטי השגיאה מגיעים מ-GetLastError() (מודפס עם %lu) — במקום perror. הדגלים ממלאים את תפקיד O_RDONLY / O_CREAT / O_TRUNC: למקור — GENERIC_READ, FILE_SHARE_READ, OPEN_EXISTING; ליעד — GENERIC_WRITE, 0 (ללא שיתוף) ו-CREATE_ALWAYS. וגם כאן: כשל בפתיחת היעד ⇐ CloseHandle(hSource) לפני היציאה.

הלולאה והסגירה

    /* Read and Write loop using Windows API */
    while (ReadFile(hSource, buffer, BUFFER_SIZE, &bytesRead, NULL) && bytesRead > 0) {
        if (!WriteFile(hDest, buffer, bytesRead, &bytesWritten, NULL) || bytesWritten != bytesRead) {
            printf("Error writing to file. Error: %lu\n", GetLastError());
            break;
        }
    }

    /* Close handles to release kernel resources */
    CloseHandle(hSource);
    CloseHandle(hDest);

    return 0;
}

הבדל סגנוני מעניין: ReadFile לא מחזירה את מספר הבייטים — היא מחזירה הצלחה/כישלון, ואת הספירה היא ממלאת דרך מצביע (&bytesRead, מטיפוס DWORD). ובמקרה של שגיאת כתיבה הקוד עושה break — יוצא מהלולאה וממשיך אל CloseHandle, כך שה-handles משתחררים בכל מקרה ("Close handles to release kernel resources").

מקור: simple_cp.pdf (עמ' 4–5)

גרסת ה-stdio היא ספריית C סטנדרטית וניידת — אותו קוד מתקמפל בלינוקס ובחלונות. גרסת ה-syscalls מדברת ישירות בשפת הקרנל של לינוקס, וגרסת ה-Windows בשפת הקרנל של חלונות — כל אחת מהשתיים נעולה לפלטפורמה שלה.

העשרה — מי בסוף קורא למי? (stdio מעל syscalls)

fopen / fread אינן קסם ואינן דרך "שלישית" להגיע לדיסק: בלינוקס הן ממומשות בתוך glibc, מחזיקות חוצץ פנימי ב-user space (Buffered I/O), וכשהחוצץ צריך מילוי או ריקון הן יורדות בעצמן — דרך wrapper routines, בדיוק כמו שראינו בשיעור הקודם — אל אותן קריאות מערכת read / write. כלומר: כל שלוש הגרסאות מגיעות בסוף לקרנל; ההבדל הוא רק כמה שכבות נוחות יש בדרך.

התוכנית שלך — main()
↓ fread(buffer, 1, 4096, source_file)
stdio בתוך glibc — חוצץ פנימי ב-User Space
↓ read(fd, buffer, n) — קריאת מערכת
Kernel — מערכת הקבצים והדיסק

גרסת ה-stdio לא עוקפת את הקרנל — היא רק מוסיפה שכבת חציצה מעליו.

היתרון של החציצה: פחות מעברים לקרנל (כל קריאת מערכת עולה מעבר בין מצבי ריצה), והקוד נייד. היתרון של syscalls ישירים: שליטה מלאה ומדויקת במה שקורה — ולכן זו השפה שבה כתובות תוכנות מערכת.

5. מלכודות למבחן

read() לא מבטיחה חוצץ מלא. הלולאה while ((bytes_read = read(...)) > 0) בנויה בדיוק בגלל זה: הקריאה האחרונה כמעט תמיד תחזיר פחות מ-4096 (שארית הקובץ), ולכן שורת הכתיבה מעבירה bytes_read — מה שנקרא בפועל — ולא BUFFER_SIZE. כתיבה של 4096 קבוע הייתה מעתיקה זבל לסוף הקובץ. לנושא Partial I/O נחזור לעומק במודול 6.

לכל API ערך כישלון אחר — אל תערבבו. fopen ⇐ NULL; open ⇐ -1; CreateFileA ⇐ INVALID_HANDLE_VALUE. בדיקה של open() מול NULL היא באג קלאסי בשאלות "מה פלט הקוד". וגם כל כתיבה נבדקת: fwrite(...) != bytes_read או bytes_written != bytes_read.

דליפת descriptors היא באג קלאסי. אם פתיחת היעד נכשלת אחרי שהמקור כבר נפתח — כל שלוש הגרסאות סוגרות קודם את המקור (fclose / close / CloseHandle) ורק אז יוצאות. close() היא ההודעה לקרנל שהוא רשאי לשחרר את המשאבים — בלעדיה הם נשארים תפוסים.

ssize_t ולא size_t. הערך המוחזר מ-read / write הוא ssize_t — עם סימן; ב-fread המונה הוא size_t. שימו לב גם שתנאי הלולאה הוא > 0 — הלולאה נעצרת גם על 0 (סוף קובץ) וגם על ערך שלילי.

6. מאחורי הקלעים של Bash — מ-ENTER ועד exec

עד עכשיו כתבנו בעצמנו תוכנית שקוראת לקריאות מערכת. אבל מה קורה ברגע שמריצים אותה? דף העזר של המרצה עוקב אחרי כל מה שמבצעת מעטפת ה-Bash מהרגע שהמשתמש מקליד ./example בטרמינל ולוחץ ENTER — חמישה שלבים ממוספרים, שסדרם המדויק הוא חומר חובה למבחן.

1

קליטת שורת הפקודה וניהול המסוף — Terminal Line Discipline

המשתמש מקליד את התווים . / e x a m p l e על המקלדת. אותות החומרה עוברים דרך מנהל ההתקן של הטרמינל — TTY Driver — שיושב בקרנל ומבצע Echo: מציג את התווים חזרה על מסך הטרמינל. התווים נאספים ונשמרים בבאפר קלט (Input Buffer) עד שהמשתמש לוחץ ENTER (המועבר כתו Newline / \n) — רק אז הטרמינל מעביר את השליטה המלאה לתהליך ה-Bash שממתין לקלט.

2

פענוח וניתוח שורת הפקודה — Tokenization & Command Line Parsing

ה-Bash שולף את המחרוזת מהבאפר ומנתח אותה (Lexical Analysis). הוא מזהה את התווים ./ בתחילת המחרוזת — וזה מלמד אותו מיד שמדובר בנתיב יחסי (Relative Path) שמצביע על קובץ הרצה בתיקיית העבודה הנוכחית (system programming), ולא בפקודה גלובלית שצריך לחפש במשתנה הסביבה PATH.

3

בדיקה האם מדובר בפקודה פנימית — Built-in Command Check

ה-Bash סורק את רשימת הפקודות הפנימיות שלו (Built-ins כמו cd, echo, export, pwd). מחרוזת שמתחילה בתווי הניווט ./ אינה תואמת אף פקודה פנימית — ולכן ה-Bash מסיק בוודאות שמדובר בפקודה חיצונית (External Executable File) שמחייבת יצירת תהליך חדש והרצה ממערכת הקבצים.

4

שכפול התהליך — קריאת המערכת fork()

ה-Bash מפעיל את קריאת המערכת fork(). מה קורה בקרנל:

  • הקרנל מייצר עותק מלא (Clone) של מבנה הנתונים task_struct של תהליך ה-Bash הנוכחי.
  • נוצר תהליך חדש (Child Process) — עותק מדויק של ה-Bash — שמקבל מזהה תהליך ייחודי משלו (PID).
  • טבלת ה-file descriptors (fd_array) ומצבי הזיכרון משוכפלים.
  • fork() מחזירה את הערך 0 לתהליך הבן, ואת ה-PID של הבן אל תהליך האב (ה-Bash המקורי) — שאלת מבחן קלאסית: מי מקבל 0 ומי מקבל PID.
5

תהליך הבן מפעיל את קריאת המערכת execve()

תהליך הבן מזהה שהוא הבן (הערך שהוחזר לו מ-fork הוא 0), ורק אז קורא ל: 

execve("/home/izhak/system programming/example", argv, envp)

(הנתיב המלא של קובץ ההרצה שבנה המשתמש). מה קורה בקרנל בעקבות ה-execve():

  • הקרנל פותח את קובץ ההרצה example במערכת הקבצים, קורא את כותרת הקובץ (ELF Header) ומוודא שהיא תקינה.
  • הקרנל מוחק ומאפס לחלוטין את מרחב הזיכרון הישן של תהליך ה-Bash שהועתק ב-fork (משחרר את הקוד, הנתונים והערימה הישנים).
  • הקרנל מקצה מרחב זיכרון חדש לגמרי, טוען אליו את הקטעים הבינאריים של התוכנית example, מגדיר את מחסנית הריצה (Stack) עם הארגומנטים ומשתני הסביבה, ומאפס את מצביע הפקודות (Instruction Pointer) להצביע על נקודת ההתחלה (main או כניסת ה-ELF) של התוכנית החדשה.

מכאן ואילך, תהליך ה-Bash המקורי נשאר ברקע במצב המתנה (Sleep / Wait), ותהליך ה-example עצמו מתחיל לרוץ במרחב המשתמש.
